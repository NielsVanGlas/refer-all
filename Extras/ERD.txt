  * id : VARCHAR(36) <<PK>>
  --
  enabled : BOOLEAN
  password : TEXT
  role : VARCHAR
  first_name : TEXT <<encrypted>>

  gender : VARCHAR
  born_date : DATE
  birth_city : TEXT <<encrypted>>
  birth_province_code : TEXT <<encrypted>>
  birth_zip_code : VARCHAR
  tax_code : TEXT <<encrypted,unique>>

  marketing_consensus : BOOLEAN
  service_terms_and_conditions : BOOLEAN
  document_type : VARCHAR
  document_id : TEXT <<encrypted>>
  created_at : TIMESTAMP
  updated_at : TIMESTAMP
}

entity "address" <<@Entity>> {
  * id : VARCHAR(36) <<PK>>
  --
  address : TEXT <<encrypted>>
  city : TEXT <<encrypted>>
  zip_code : TEXT <<encrypted>>
  province_code : TEXT <<encrypted>>
  country_code : TEXT <<encrypted>>
  created_at : TIMESTAMP
  updated_at : TIMESTAMP
}

  * id : VARCHAR(36) <<PK>>
  --
  blood_type : VARCHAR
  allergies : TEXT <<encrypted>>
  chronic_conditions : TEXT <<encrypted>>
  created_at : TIMESTAMP
  updated_at : TIMESTAMP
}

entity "doctor_profile" <<@Entity>> {
  * id : VARCHAR(36) <<PK>>
  --
  license_number : TEXT <<encrypted,unique>>
  created_at : TIMESTAMP
  updated_at : TIMESTAMP
}

entity "specialization" <<@Entity>> {
  * id : VARCHAR(36) <<PK>>
  --
  specialization : TEXT <<encrypted,unique>>
  created_at : TIMESTAMP
  updated_at : TIMESTAMP
}
